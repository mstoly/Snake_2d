cmake_minimum_required (VERSION 3.15)

project ("Snake_2d" LANGUAGES CXX)

set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

add_executable (Snake_2d "src/main.cpp"
						 "src/Mesh.cpp"
                         "src/Model.cpp"
                         "src/Renderer.cpp"
						 "src/ShaderProgram.cpp")

target_include_directories(Snake_2d PRIVATE include)
target_compile_features(Snake_2d PRIVATE cxx_std_20)

include(FetchContent)

FetchContent_Declare(
	glm
	GIT_REPOSITORY	https://github.com/g-truc/glm.git
	GIT_TAG 	    1.0.3
)

FetchContent_Declare(
	glfw
	GIT_REPOSITORY	https://github.com/glfw/glfw.git
	GIT_TAG 	    3.4
)

FetchContent_MakeAvailable(glm glfw)

add_subdirectory(external/glad)

target_link_libraries(Snake_2d PRIVATE
	glm::glm
	glfw
	glad)

file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/shaders" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
